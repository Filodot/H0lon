% ---------------------------------------------------------------- wide displays
% A display formula cannot be broken: wider than a (narrow) column, it would run into the
% neighbouring column. Such a formula is scaled down to the column width; the others are
% set as usual by the original \[ ... \]. (Delimited by \], so a formula with \tag or an
% alignment environment of its own is not meant for this; Pandoc writes plain \[ ... \].)
\makeatletter
\newsavebox{\hz@mbox}
\long\def\hz@display#1\]{%
  \sbox{\hz@mbox}{\(\displaystyle #1\)}%
  \ifdim\wd\hz@mbox>\linewidth
    \par\addvspace{\abovedisplayskip}\noindent
    \hfil\resizebox{\linewidth}{!}{\usebox{\hz@mbox}}\hfil\par
    \addvspace{\belowdisplayskip}\noindent\ignorespaces
  \else
    \hz@opendisplay #1\hz@closedisplay
  \fi}
% \[ is a robust command: its body lives in the control sequence named "[ " (with a space),
% which DeclareRobustCommand would overwrite, so the original body is copied first.
\AtBeginDocument{%
  \expandafter\let\expandafter\hz@opendisplay\csname [ \endcsname
  \let\hz@closedisplay\]%
  \DeclareRobustCommand{\[}{\relax\ifmmode\@badmath\else\expandafter\hz@display\fi}}
\makeatother

% Run-in block: #1 counter (empty = unnumbered), #2 label, #3 mark, #4 title, #5 body font.
% \refstepcounter (the hyperref anchor) comes after \Needspace and \noindent: when the
% block moves to the next column or page, links must land there.
\newcommand{\hzruninhead}[5]{%
  \Needspace{2\baselineskip}\par\smallskip\noindent
  \ifstrempty{#1}{}{\refstepcounter{#1}}%
  #3\textbf{#2\ifstrempty{#1}{}{~\csname the#1\endcsname}.}%
  \ifstrempty{#4}{}{\hzgap{}\emph{#4.}}\hzgap{}#5\ignorespaces}

% longtable cannot work inside multicols (it takes over the page builder). A table therefore
% interrupts the columns: the text above is balanced, the table spans the full width, the
% columns go on below it. Only a table placed directly in the columns is handled; a table
% inside another environment (a list, a proof) makes XeLaTeX fail, and the HTML fallback
% engine takes over.
% Pandoc wraps a table without a caption into a brace group, { \def\LTcaptype{none} ... },
% and \end{multicols} cannot be given inside it: that group is left first (and opened again,
% with \LTcaptype restored, for the closing brace that Pandoc writes after the table); the
% columns are entered again after that closing brace (\aftergroup).
% The order matters: \makesavenoteenv (footnotes in tables) opens its own group in a
% before-begin hook, so the columns must be left by a hook that is registered before it
% and entered again by one that is registered after it.
\makeatletter
\newcount\hz@interrupted % 0: columns not interrupted, 1: left directly, 2: left from a { group
\newcommand{\hz@reenter}{\begin{multicols}{2}}
\BeforeBeginEnvironment{longtable}{%
  \ifdefstring{\@currenvir}{multicols}{%
    \ifnum\currentgrouptype=1 % a { group
      \xdef\hz@ltcaptype{\noexpand\def\noexpand\LTcaptype{\LTcaptype}}%
      \egroup
      \end{multicols}%
      \bgroup
      \hz@ltcaptype
      \global\hz@interrupted=2
    \else
      \end{multicols}%
      \global\hz@interrupted=1
    \fi}{}}
\makeatother
% Allow footnotes in longtable head/foot

\makesavenoteenv{longtable}
\makeatletter
\AfterEndEnvironment{longtable}{%
  \ifcase\hz@interrupted
  \or \global\hz@interrupted=0 \begin{multicols}{2}%
  \or \global\hz@interrupted=0 \aftergroup\hz@reenter
  \fi}
\makeatother
