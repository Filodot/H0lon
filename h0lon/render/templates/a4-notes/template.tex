% Run-in block: #1 counter (empty = unnumbered), #2 label, #3 mark, #4 title, #5 body font.
% \refstepcounter (the hyperref anchor) comes after \Needspace and \noindent: when the
% block moves to the next page, links must land there and not at the previous page's foot.
\newcommand{\hzruninhead}[5]{%
  \Needspace{3\baselineskip}\par\medskip\noindent
  \ifstrempty{#1}{}{\refstepcounter{#1}}%
  #3\textbf{#2\ifstrempty{#1}{}{~\csname the#1\endcsname}.}%
  \ifstrempty{#4}{}{\enspace\emph{#4.}}\enspace#5\ignorespaces}

\newenvironment{hzdefinition}[1][]
  {\hzruninhead{hzdefinition}{Определение}{\hzmark}{#1}{}}
  {\par\medskip}
\newenvironment{hzproblem}[1][]
  {\hzruninhead{hzproblem}{Задача}{\hzmark}{#1}{}}
  {\par\medskip}
\newenvironment{hzexample}[1][]
  {\hzruninhead{hzexample}{Пример}{}{#1}{}}
  {\par\medskip}

% Statements: bold numbered label, italic text.
\newcommand{\hzstatementhead}[3]{%
  \Needspace{3\baselineskip}\par\medskip\noindent\refstepcounter{#1}%
  \textbf{#2~\csname the#1\endcsname.}%
  \ifstrempty{#3}{}{\enspace\textbf{(#3)}}\enspace\itshape\ignorespaces}
\newenvironment{hzlemma}[1][]
  {\hzstatementhead{hzlemma}{Лемма}{#1}}
  {\par\medskip}
\newenvironment{hzproposition}[1][]
  {\hzstatementhead{hzproposition}{Предложение}{#1}}
  {\par\medskip}
\newenvironment{hzcorollary}[1][]
  {\hzstatementhead{hzcorollary}{Следствие}{#1}}
  {\par\medskip}

% Theorem: label column on the left (breaks across pages like any list item).
% The anchor (\refstepcounter, vertical mode) follows \Needspace: if the theorem moves to the
% next page, the anchor moves with it.
\newenvironment{hztheorem}[1][]
  {\Needspace{4\baselineskip}\par\medskip\refstepcounter{hztheorem}%
   \begin{description}[style=multiline,leftmargin=36mm,labelwidth=31mm,labelsep=5mm,
                       itemsep=0pt,topsep=0pt,partopsep=0pt,font=\normalfont]
   \item[\textbf{Теорема~\thehztheorem.}\ifstrempty{#1}{}{\newline\textbf{(#1)}}]%
   \itshape\ignorespaces}
  {\end{description}\par\medskip}
